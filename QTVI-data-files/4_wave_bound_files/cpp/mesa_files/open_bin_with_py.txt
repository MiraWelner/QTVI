import os
import struct


# --- Converted Code Structure ---
class ComparisonData:
    def __init__(self):
        self.ecgRIndex = []
        self.ppgMaxAmps = []
        self.ppgMinAmps = []
        self.ppgSignal = []
        self.pairs = []


def readVecFromBin(file, vec):
    data = file.read(8)
    if not data or len(data) < 8:
        return
    count = struct.unpack("<Q", data)[0]
    for _ in range(count):
        val_data = file.read(8)
        if len(val_data) == 8:
            vec.append(float(struct.unpack("<q", val_data)[0]))


def readDoubleVecFromBin(file, vec):
    data = file.read(8)
    if not data or len(data) < 8:
        return
    count = struct.unpack("<Q", data)[0]
    if count > 0:
        raw_bytes = file.read(count * 8)
        if len(raw_bytes) == count * 8:
            vec.extend(struct.unpack(f"<{count}d", raw_bytes))


def readAllFromBin(path):
    results = []
    if not os.path.exists(path):
        print(f"Error: File not found at {path}")
        return results

    with open(path, "rb") as file:
        data = file.read(8)
        if not data or len(data) < 8:
            return results
        numBins = struct.unpack("<Q", data)[0]

        for _ in range(numBins):
            entry = ComparisonData()
            readVecFromBin(file, entry.ecgRIndex)
            readVecFromBin(file, entry.ppgMaxAmps)
            readVecFromBin(file, entry.ppgMinAmps)
            readDoubleVecFromBin(file, entry.ppgSignal)

            pair_count_data = file.read(8)
            if pair_count_data and len(pair_count_data) == 8:
                numPairs = struct.unpack("<Q", pair_count_data)[0]
                for _ in range(numPairs):
                    p_data = file.read(8)
                    e_data = file.read(8)
                    if len(p_data) == 8 and len(e_data) == 8:
                        entry.pairs.append(
                            (
                                float(struct.unpack("<q", p_data)[0]),
                                float(struct.unpack("<q", e_data)[0]),
                            )
                        )
            results.append(entry)
    return results
